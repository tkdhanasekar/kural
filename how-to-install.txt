vim /etc/nginx/conf.d/kural.conf
server {
    #listen *:443;
    server_name akural.hashlabs.in;

    location /userpdfs {
        alias /root/code/kuralweb/userpdfs;
    }

    location /static/ {
        root /root/code/kuralweb;
    }

   location / {
        include proxy_params;
        proxy_pass http://unix:/run/gunicorn.sock;
    }


    listen 443 ssl; # managed by Certbot
    ssl_certificate /etc/letsencrypt/live/akural.hashlabs.in/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/akural.hashlabs.in/privkey.pem; # managed by Certbot
    include /etc/letsencrypt/options-ssl-nginx.conf; # managed by Certbot
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem; # managed by Certbot

}
server {
    if ($host = akural.hashlabs.in) {
        return 301 https://$host$request_uri;
    } # managed by Certbot


    server_name akural.hashlabs.in;
    listen 80;
    return 404; # managed by Certbot


}

:wq!

systemctl restart nginx.service

vim /etc/systemd/system/gunicorn.service
[Unit]
Description=gunicorn daemon
Requires=gunicorn.socket
After=network.target

[Service]
User=root
Group=root
WorkingDirectory=/root/code/
EnvironmentFile=/root/code/.env

ExecStart= /root/code/venv/bin/gunicorn \
            --access-logfile - \
            --workers 3 \
            --timeout 500 \
            --bind unix:/run/gunicorn.sock \
            tirukkuralweb.wsgi:application
TimeoutStopSec=500
[Install]
WantedBy=multi-user.target
:wq!

vim /etc/systemd/system/gunicorn.socket
[Unit]
Description=gunicorn socket

[Socket]
ListenStream=/run/gunicorn.sock

[Install]
WantedBy=sockets.target
:wq!

# dameon reload
systemctl daemon-reload

# Enable guncorn and socket
